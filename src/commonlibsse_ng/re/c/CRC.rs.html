36/// Calculate CRC32 from bytes.
37#[inline]
38pub const fn generate_crc32(data: &[u8]) -> u32 {
39    let mut hash = 0;
40    let mut i = 0;
41
42    while i < data.len() {
43        let index = (hash ^ (data[i] as u32)) & 0xFF;
44        hash = (hash >> 8) ^ CRC32_TABLE[index as usize];
45        i += 1;
46    }
47    hash
48}

50/// Calculate CRC32 from [`u32`].
51#[inline]
52pub const fn crc32_from_u32(input: u32) -> u32 {
53    generate_crc32(&input.to_le_bytes())
54}
